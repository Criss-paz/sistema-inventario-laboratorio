-- ===== CP3-10: no se puede devolver mas de lo que salio (RN-22) =====
DO $$ DECLARE v INT; s INT; l INT;
BEGIN
  SELECT id_movimiento INTO s FROM movimiento WHERE observacion='Prueba CP3-09 salida equivocada';
  SELECT id_lote INTO l FROM detalle_movimiento WHERE id_movimiento=s LIMIT 1;
  CALL sp_registrar_devolucion(1, s, 'Intento de devolver 99 de las 12 que salieron', ARRAY[l], ARRAY[99::numeric], v);
EXCEPTION WHEN OTHERS THEN RAISE NOTICE 'CP3-10 RECHAZADO: %', SQLERRM;
END $$;
